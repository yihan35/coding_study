def trap(height):
    """
    计算数组表示的柱子能接住的雨水量（双指针最优解）
    :param height: List[int] - 每个元素代表柱子的高度
    :return: int - 总接水量
    """
    # 边界条件：柱子数量小于3，无法形成凹槽，接水量为0
    if len(height) < 3:
        return 0
    
    # 初始化双指针和核心变量
    left = 0                  # 左指针，从最左侧开始
    right = len(height) - 1   # 右指针，从最右侧开始
    left_max = 0              # 左指针左侧（含当前）的最大高度
    right_max = 0             # 右指针右侧（含当前）的最大高度
    total_water = 0           # 总接水量
    
    # 双指针向中间遍历，直到相遇
    while left <= right:
        # 关键判断：左最大值更小，说明左指针位置的接水量由左最大值决定
        if left_max < right_max:
            # 更新左最大值（当前柱子更高，无法接水，仅更新最大值）
            left_max = max(left_max, height[left])
            # 计算当前左指针位置的接水量（负数则取0）
            current_water = left_max - height[left]
            total_water += current_water
            # 左指针右移
            left += 1
        # 右最大值更小或相等，说明右指针位置的接水量由右最大值决定
        else:
            # 更新右最大值
            right_max = max(right_max, height[right])
            # 计算当前右指针位置的接水量
            current_water = right_max - height[right]
            total_water += current_water
            # 右指针左移
            right -= 1
    
    return total_water

    height1 = [0,1,0,2,1,0,1,3,2,1,2,1]
    print(f"示例1接水量：{trap(height1)}")
    
    # 示例2：输出9

    print(f"示例2接水量：{trap(height2)}")

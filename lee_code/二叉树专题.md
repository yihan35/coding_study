# 二叉树 Hot 题单代码汇总

统一用四步心法标注：
1. 结束条件
2. 假设左右子树已完成（递归）
3. 当前层要做的事
4. 返回结果

## 公共基础设施

```python
from collections import deque


class TreeNode:
    def __init__(self, val, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


def is_null(s):
    return s == 'null' or s == 'NULL' or s == 'None' or s == '#'


def build_tree(level_order):
    if not level_order or is_null(level_order[0]):
        return None
    root = TreeNode(level_order[0])
    queue = deque([root])
    index = 1
    while queue and index < len(level_order):
        node = queue.popleft()  # 一旦弹出节点，为其分配左右孩子
        if index < len(level_order) and level_order[index] is not None:
            node.left = TreeNode(level_order[index])
            queue.append(node.left)
        index += 1
        if index < len(level_order) and level_order[index] is not None:
            node.right = TreeNode(level_order[index])
            queue.append(node.right)
        index += 1
    return root


def inorderTraversal(root):
    if not root:  # 1. 结束条件：空节点，没有值
        return []
    left = inorderTraversal(root.left)    # 2. 假设左子树已完成
    right = inorderTraversal(root.right)  # 假设右子树已完成
    return left + [root.val] + right      # 3+4. 左 + 自己 + 右，返回
```

## 1. 二叉树的最大宽度

```python
def widthOfBinaryTree(root):
    # 1. 结束条件：空树宽度为0
    if not root:
        return 0

    ans = 0
    # 给每个节点编号（满二叉树规律：左孩子 2*idx，右孩子 2*idx+1）
    queue = deque([(root, 0)])

    while queue:
        level_size = len(queue)
        _, first_idx = queue[0]  # 本层第一个节点的编号

        for _ in range(level_size):
            # 2. 每弹出一个节点，就为它的子节点分配编号（相当于"进入下一层"）
            node, idx = queue.popleft()
            if node.left:
                queue.append((node.left, idx * 2))
            if node.right:
                queue.append((node.right, idx * 2 + 1))

        # 3. 当前层要做的事：本层最后编号 - 第一个编号 + 1 = 本层宽度
        # 4. 更新全局最大值
        ans = max(ans, idx - first_idx + 1)

    return ans
```

## 2. 相同二叉树

```python
def isSameTree(p, q):
    # 1. 结束条件：都为空，相同；有一个为空或值不同，直接短路返回False
    if not p and not q:
        return True
    if not p or not q or p.val != q.val:
        return False

    # 2. 假设左右子树的"相同性"已经算出来了
    left_same = isSameTree(p.left, q.left)
    right_same = isSameTree(p.right, q.right)

    # 3+4. 当前层要做的事：左右都相同，整体才相同
    return left_same and right_same
```

## 3. 对称二叉树

```python
def isSymmetric(root):
    if not root:
        return True

    def isMirror(p, q):
        # 1. 结束条件
        if not p and not q:
            return True
        if not p or not q or p.val != q.val:
            return False

        # 2. 假设子树已完成（注意是"交叉"递归：p的左对q的右，p的右对q的左）
        outer = isMirror(p.left, q.right)
        inner = isMirror(p.right, q.left)

        # 3+4. 当前层：两边都对称，才算对称
        return outer and inner

    return isMirror(root.left, root.right)
```

## 4. 二叉树展开为链表

```python
def flatten(root):
    # 1. 结束条件：空节点，没有链可展开
    if not root:
        return None

    # 2. 假设左右子树已经各自被展开成链表，拿到各自链表的"尾节点"
    left_tail = flatten(root.left)
    right_tail = flatten(root.right)

    # 3. 当前层要做的事：把左链表接到右链表前面，再把整个左链表挂到root.right
    if root.left:
        left_tail.right = root.right
        root.right = root.left
        root.left = None

    # 4. 返回"以root为头的这条链"的尾节点，给上一层用
    tail = right_tail or left_tail or root
    return tail
```

## 5. 翻转二叉树

```python
def invertTree(root):
    # 1. 结束条件：空节点，没什么可翻的
    if not root:
        return None

    # 2. 假设左右子树都已经翻转好了
    left = invertTree(root.left)
    right = invertTree(root.right)

    # 3. 当前层要做的事：把翻转好的子树左右互换，挂到自己身上
    root.left = right
    root.right = left

    # 4. 返回翻转好的"自己"
    return root
```

## 6. 平衡二叉树

```python
def isBalanced(root):
    def height(node):
        # 1. 结束条件：空节点，高度为0
        if not node:
            return 0

        # 2. 假设左右子树高度已算出（同时用-1做"不平衡"的短路信号）
        left = height(node.left)
        if left == -1:
            return -1  # 左边已经不平衡，直接短路，不用再算右边
        right = height(node.right)
        if right == -1:
            return -1

        # 3. 当前层要做的事：判断左右高度差是否超过1（本层的判断条件）
        if abs(left - right) > 1:
            return -1

        # 4. 条件都满足，正常返回当前节点的高度
        return max(left, right) + 1

    return height(root) != -1
```

## 7. 二叉树的直径

```python
def diameterOfBinaryTree(root):
    ans = [0]  # 全局变量：用列表包一层，方便在内层函数里直接修改

    def depth(node):
        # 1. 结束条件：空节点，深度为0
        if not node:
            return 0

        # 2. 假设左右子树深度已经算出
        left = depth(node.left)
        right = depth(node.right)

        # 3. 当前层要做的事：更新全局最大值（经过当前节点的路径长度 = 左深度+右深度）
        ans[0] = max(ans[0], left + right)

        # 4. 返回给上一层的是"深度"，不是"直径"（两者的return目的不一样）
        return max(left, right) + 1

    depth(root)
    return ans[0]
```

## 8. 二叉树中的最大路径和

```python
def maxPathSum(root):
    ans = [float('-inf')]  # 全局变量：记录目前为止最大的路径和

    def gain(node):
        # 1. 结束条件：空节点，贡献为0
        if not node:
            return 0

        # 2. 假设左右子树"能贡献给父节点的最大值"已经算出
        # 负贡献直接舍弃（相当于不选这条边），用max(...,0)体现
        left_gain = max(gain(node.left), 0)
        right_gain = max(gain(node.right), 0)

        # 3. 当前层要做的事：更新全局最大路径和（这条路径可以同时用左右两边）
        ans[0] = max(ans[0], node.val + left_gain + right_gain)

        # 4. 但return给上一层时，只能选一边往上"接"（一条链不能分叉）
        return node.val + max(left_gain, right_gain)

    gain(root)
    return ans[0]
```

## 9. 二叉树的右视图

```python
def rightSideView(root):
    ans = []  # 全局变量：存放最终结果

    def dfs(node, depth):
        # 1. 结束条件：空节点，不做任何事
        if not node:
            return

        # 2+3. 当前层要做的事：每层第一次访问到（depth==len(ans)）就是最右节点
        if depth == len(ans):
            ans.append(node.val)

        # 先右后左，保证每层"第一次到达"的一定是最右边的节点
        dfs(node.right, depth + 1)
        dfs(node.left, depth + 1)

    dfs(root, 0)
    return ans
```

## 10. 验证二叉搜索树

```python
def isValidBST(root):
    def valid(node, lower, upper):
        # 1. 结束条件：空节点，天然合法
        if not node:
            return True

        # 2. 当前层要做的事（本质条件）：值必须落在(lower, upper)开区间内
        if not (lower < node.val < upper):
            return False

        # 3+4. 递归左右子树，同时收紧边界：
        # 左子树的上界变成当前值；右子树的下界变成当前值
        return valid(node.left, lower, node.val) and valid(node.right, node.val, upper)

    return valid(root, float('-inf'), float('inf'))
```

## 11. 判断子结构

```python
def isSubStructure(A, B):
    # 1. 结束条件：A或B任一为空，直接不算子结构
    if not A or not B:
        return False

    def contains(a, b):
        # 内层判断：以a为起点，是否完整包含b
        if not b:
            return True  # b已经走完了，说明匹配上了
        if not a or a.val != b.val:
            return False
        return contains(a.left, b.left) and contains(a.right, b.right)

    # 2+3+4. 当前节点直接试一次；不行的话，去A的左子树、右子树分别再试
    return contains(A, B) or isSubStructure(A.left, B) or isSubStructure(A.right, B)
```

## 12. 二叉搜索树的最小绝对差

```python
def getMinimumDifference(root):
    ans = [float('inf')]  # 全局变量：记录最小差值
    prev = [None]          # 全局变量：记录中序遍历中"上一个"访问的值

    def inorder(node):
        # 1. 结束条件
        if not node:
            return

        # 2. 先处理左子树（中序：左->根->右）
        inorder(node.left)

        # 3. 当前层要做的事：BST中序遍历是递增序列，最小差只会出现在相邻两个数之间
        if prev[0] is not None:
            ans[0] = min(ans[0], node.val - prev[0])
        prev[0] = node.val

        # 4. 再处理右子树
        inorder(node.right)

    inorder(root)
    return ans[0]
```

## 13. 二叉搜索树中第K小的元素

```python
def kthSmallest(root, k):
    # 用栈模拟中序遍历（不用递归，迭代版本）
    stack = []
    node = root
    while stack or node:
        while node:            # 一路往左压栈，走到最左（最小值）
            stack.append(node)
            node = node.left
        node = stack.pop()      # 弹出的顺序天然是从小到大
        k -= 1
        if k == 0:              # 第k次弹出，就是第k小
            return node.val
        node = node.right       # 转向右子树
```

## 14. 从前序与中序遍历序列构造二叉树

```python
def buildTree_pre_in(preorder, inorder):
    # 1. 结束条件：序列为空，没有树可建
    if not preorder:
        return None

    # 2. 当前层要做的事：前序第一个永远是"根"
    root_val = preorder[0]
    root = TreeNode(root_val)

    # 在中序序列里找到根的位置：根左边是左子树，根右边是右子树
    idx = inorder.index(root_val)

    # 3. 用这个位置切分前序、中序序列，分别递归建左右子树
    root.left = buildTree_pre_in(preorder[1:idx + 1], inorder[:idx])
    root.right = buildTree_pre_in(preorder[idx + 1:], inorder[idx + 1:])

    # 4. 返回建好的根节点
    return root
```

## 15. 从中序与后序遍历序列构造二叉树

```python
def buildTree_in_post(inorder, postorder):
    # 1. 结束条件：序列为空
    if not postorder:
        return None

    # 2. 当前层要做的事：后序最后一个永远是"根"
    root_val = postorder[-1]
    root = TreeNode(root_val)

    idx = inorder.index(root_val)

    # 3. 切分中序、后序序列，分别递归建左右子树
    root.left = buildTree_in_post(inorder[:idx], postorder[:idx])
    root.right = buildTree_in_post(inorder[idx + 1:], postorder[idx:-1])

    # 4. 返回建好的根节点
    return root
```

## 16. 二叉树的最近公共祖先（LCA）

```python
def lowestCommonAncestor(root, p, q):
    # 1. 结束条件：走到空，或者当前节点本身就是p/q之一，直接返回
    if not root or root == p or root == q:
        return root

    # 2. 假设左右子树各自去找p、q，能找到就返回对应节点
    left = lowestCommonAncestor(root.left, p, q)
    right = lowestCommonAncestor(root.right, p, q)

    # 3. 当前层要做的事：左右都非空，说明p、q分别在左右两边，当前节点就是分岔点(LCA)
    if left and right:
        return root

    # 4. 只有一边找到，说明p、q都在那一边，把结果继续往上传
    return left or right
```

## 17. 二叉搜索树的最近公共祖先

```python
def lowestCommonAncestorBST(root, p, q):
    # 利用BST"有序"的性质，不需要两边都递归，直接按值大小判断往哪边走

    # 1+2. p、q都比当前节点小，说明LCA在左子树
    if p.val < root.val and q.val < root.val:
        return lowestCommonAncestorBST(root.left, p, q)

    # p、q都比当前节点大，说明LCA在右子树
    if p.val > root.val and q.val > root.val:
        return lowestCommonAncestorBST(root.right, p, q)

    # 3+4. 一个偏左一个偏右（或有一个正好等于root），当前节点就是分岔点
    return root
```

## 18. 二叉树的层序遍历

```python
def levelOrder(root):
    if not root:
        return []

    ans = []
    queue = deque([root])

    while queue:
        level_size = len(queue)  # 记录"这一层"有多少个节点，作为循环边界
        level_vals = []

        for _ in range(level_size):
            node = queue.popleft()          # 弹出：轮到这个节点被处理
            level_vals.append(node.val)
            if node.left:
                queue.append(node.left)      # 孩子入队，等下一层再被弹出
            if node.right:
                queue.append(node.right)

        ans.append(level_vals)  # 一层处理完，整体加入结果

    return ans
```

## 19. 找树左下角的值（513）

```python
def findBottomLeftValue(root):
    queue = deque([root])
    ans = root.val

    while queue:
        level_size = len(queue)
        for i in range(level_size):
            node = queue.popleft()
            if i == 0:                       # 每层第一个弹出的，就是这一层最左边的节点
                ans = node.val
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)

    # 循环走到最后一层才停止，所以ans最终记录的是"最后一层的最左节点"
    return ans
```

## 20. 将有序数组转化为二叉搜索树

```python
def sortedArrayToBST(nums):
    # 1. 结束条件：数组为空，没有节点可建
    if not nums:
        return None

    # 2. 当前层要做的事：取数组中点当根（保证左右子树节点数尽量平衡）
    mid = len(nums) // 2
    root = TreeNode(nums[mid])

    # 3. 用中点切分数组，左半部分递归建左子树，右半部分递归建右子树
    root.left = sortedArrayToBST(nums[:mid])
    root.right = sortedArrayToBST(nums[mid + 1:])

    # 4. 返回建好的根节点
    return root
```
